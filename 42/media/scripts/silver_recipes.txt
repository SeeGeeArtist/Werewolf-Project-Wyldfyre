    /* 0.4.62: silver nitrate is a FLUID (silver_fluids.txt), in two stages.

       Stage 1, anywhere with a surface: nitrate (NPK fertilizer, TWO USES of
       one bag; a bag holds eight; or, 0.4.65, ONE instant cold pack, whose
       filling is ammonium nitrate - `1:Base.Coldpack` is the per-item
       amount syntax), the sulphuric acid of a cracked car battery
       (destroyed) and one Silver Fragments go into an EMPTY GLASS fluid
       container, which is KEPT and filled with up to 1.0 L of Raw Silver
       Nitrate (jar and beer bottle hold 1.0 L, a drinking glass 0.2 L; what
       does not fit is lost). The opening tool (knife, screwdriver, pliers) is
       kept. No skill. Nothing plastic: plastic never carries the Glass tag,
       and OnTestNitrateInput additionally refuses Glass-tagged items that are
       not fluid containers (windshields, shards).

       Stage 2 is not a recipe: put the vessel INTO a lit campfire, barbecue
       or stove like a pot of tainted water. NitrateHeatPatch turns the raw
       mix into Silver Nitrate at the tainted-water boil rate (container
       temperature > 1.5; a lit fire is 1.8). The jar does not have to be a
       cooking pot: the patch reads the fire's temperature, not the item's. */
    craftRecipe WTAGarou_PrepareSilverNitrate
    {
        Time = 300,
        Tags = AnySurfaceCraft,
        timedAction = Making,
        category = Weaponry,
        AllowBatchCraft = False,
        OnTest = WTAGarou.SilverAmmo.OnTestNitrateInput,
        inputs
        {
            item 1 tags[Glass;GlassBottle] mode:keep flags[ItemIsFluid;IsEmpty],
            +fluid 1.0 WTAGarou_SilverNitrateRaw,
            item 1 [Base.SilverScrap],
            item 2 [Base.Fertilizer;1:Base.Coldpack],
            item 1 [Base.CarBattery1;Base.CarBattery2;Base.CarBattery3] mode:destroy,
            item 1 tags[SharpKnife;Screwdriver;Pliers;MetalworkingPliers] mode:keep flags[Prop1;MayDegradeLight],
        }
        outputs
        {
        }
    }
